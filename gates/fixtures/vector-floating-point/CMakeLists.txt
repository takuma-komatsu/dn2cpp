cmake_minimum_required(VERSION 3.20)
project(dn2cpp_vector_fma CXX)

get_filename_component(DN2CPP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(DN2CPP_VECTOR_FMA_ARCH none CACHE STRING "Host-supported native FMA target: none, avx2, or arm64")

function(vector_fma_target target expected)
    add_executable(${target} fma.cpp)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${DN2CPP_ROOT}/runtime/core" "${DN2CPP_ROOT}/third_party/highway")
    target_compile_definitions(${target} PRIVATE DN2CPP_USE_HIGHWAY DN2CPP_VECTOR_EXPECT_FMA=${expected})
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -ffp-contract=off)
    endif()
endfunction()

vector_fma_target(vector-fma-software 0)
# Highway falls back to its scalar target on compilers with a known EMU128 bug.
target_compile_definitions(vector-fma-software PRIVATE HWY_COMPILE_ONLY_EMU128)

if(NOT DN2CPP_VECTOR_FMA_ARCH STREQUAL "none")
    vector_fma_target(vector-fma-native 1)
    if(DN2CPP_VECTOR_FMA_ARCH STREQUAL "avx2")
        target_compile_definitions(vector-fma-native PRIVATE HWY_DISABLE_PCLMUL_AES HWY_DISABLE_F16C)
        if(MSVC)
            target_compile_options(vector-fma-native PRIVATE /arch:AVX2)
        else()
            target_compile_options(vector-fma-native PRIVATE -mavx2 -mfma -mbmi2)
        endif()
    elseif(NOT DN2CPP_VECTOR_FMA_ARCH STREQUAL "arm64")
        message(FATAL_ERROR "Unknown native FMA target: ${DN2CPP_VECTOR_FMA_ARCH}")
    endif()
endif()
